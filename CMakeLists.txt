# Modular graphics subsystem: Vulkan is preferred, D3D only a reserved ABI slot.
option(GZ_ENABLE_VULKAN "Try to compile Vulkan backend when the SDK is available" ON)
option(GZ_REQUIRE_VULKAN "Fail configuration when Vulkan SDK is unavailable" OFF)
if(GZ_REQUIRE_VULKAN AND NOT GZ_ENABLE_VULKAN)
  message(FATAL_ERROR "GZ_REQUIRE_VULKAN=ON is incompatible with GZ_ENABLE_VULKAN=OFF")
endif()
if(GZ_ENABLE_VULKAN)
  find_package(Vulkan QUIET)
endif()
add_library(gzrender STATIC src/RenderBackend.cpp src/RenderGraph.cpp src/GeometryStaging.cpp src/DirectXCompatStub.cpp)
target_include_directories(gzrender PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
if(Vulkan_FOUND)
  target_sources(gzrender PRIVATE src/VulkanBackend.cpp src/VulkanScenePass.cpp)
  target_link_libraries(gzrender PUBLIC Vulkan::Vulkan)
  target_compile_definitions(gzrender PUBLIC GZ_VULKAN_COMPILED=1)
  message(STATUS "GRAY ZONE: Vulkan backend enabled with Vulkan SDK")
  find_program(GZ_GLSLANG_VALIDATOR NAMES glslangValidator glslangValidator.exe
      HINTS "$ENV{VULKAN_SDK}/Bin" "$ENV{VULKAN_SDK}/Bin32" "$ENV{VULKAN_SDK}/bin")
  if(GZ_GLSLANG_VALIDATOR)
    set(GZ_GPU_SHADERS)
    foreach(stage IN ITEMS vert frag)
      set(shader_src "${CMAKE_CURRENT_SOURCE_DIR}/assets/shaders/vulkan/native_scene.${stage}")
      set(shader_spv "${CMAKE_CURRENT_SOURCE_DIR}/assets/shaders/vulkan/native_scene.${stage}.spv")
      add_custom_command(OUTPUT "${shader_spv}"
          COMMAND "${GZ_GLSLANG_VALIDATOR}" -V "${shader_src}" -o "${shader_spv}"
          DEPENDS "${shader_src}" COMMENT "Compiling Vulkan native ${stage} shader")
      list(APPEND GZ_GPU_SHADERS "${shader_spv}")
    endforeach()
    add_custom_target(gz_vulkan_shaders ALL DEPENDS ${GZ_GPU_SHADERS})
    message(STATUS "GRAY ZONE: native GPU shaders compiled with ${GZ_GLSLANG_VALIDATOR}")
  else()
    if(GZ_REQUIRE_VULKAN)
      message(FATAL_ERROR "Native Vulkan requested, but glslangValidator is missing. Install Vulkan SDK with glslangValidator.")
    endif()
    message(WARNING "glslangValidator not found: Vulkan CPU upload works, native GPU pass needs compiled SPIR-V")
  endif()
else()
  if(GZ_REQUIRE_VULKAN)
    message(FATAL_ERROR "Vulkan SDK is required for this build. Install Vulkan SDK and re-run CMake.")
  endif()
  target_sources(gzrender PRIVATE src/VulkanUnavailable.cpp)
  target_compile_definitions(gzrender PUBLIC GZ_VULKAN_COMPILED=0)
  message(WARNING "Vulkan SDK not found. Compiling software fallback and RHI tests only.")
endif()
if(MSVC)
  target_compile_options(gzrender PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(gzrender PRIVATE -Wall -Wextra -Wpedantic)
endif()

  endif()
  target_link_libraries(GRAY_ZONE_1989 PRIVATE gzcore gzrender user32 gdi32 winmm)
  target_compile_definitions(GRAY_ZONE_1989 PRIVATE UNICODE _UNICODE)
  if(MSVC)
    target_compile_options(GRAY_ZONE_1989 PRIVATE /utf-8)
    target_link_options(GRAY_ZONE_1989 PRIVATE /ENTRY:wWinMainCRTStartup)
  endif()
endif()
